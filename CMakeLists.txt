cmake_minimum_required(VERSION 3.21)
project(Metroid VERSION 0.1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(Qt6 6.4 REQUIRED COMPONENTS Widgets)
# WIN32: a windowed program, so Send to and shortcuts open no console window.
qt_add_executable(metroid-desktop WIN32 desktop/main.cpp)
if(WIN32)
    target_sources(metroid-desktop PRIVATE desktop/metroid.rc)
endif()
target_link_libraries(metroid-desktop PRIVATE Qt6::Widgets)
qt_add_resources(metroid-desktop interface PREFIX "/" BASE assets FILES assets/chevron.png)
install(TARGETS metroid-desktop RUNTIME DESTINATION bin)
install(FILES assets/metroid.svg DESTINATION share/icons/hicolor/scalable/apps)
include(CTest)
if(BUILD_TESTING)
    qt_add_executable(metroid-wheel-test tests/radialwheel.cpp)
    target_link_libraries(metroid-wheel-test PRIVATE Qt6::Widgets)
    add_test(NAME radial-wheel COMMAND metroid-wheel-test)
    set_tests_properties(radial-wheel PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen")
    qt_add_executable(metroid-dropwheel-test tests/dropwheel.cpp)
    target_link_libraries(metroid-dropwheel-test PRIVATE Qt6::Widgets)
    add_test(NAME drop-wheel COMMAND metroid-dropwheel-test)
    set_tests_properties(drop-wheel PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen")
endif()
