cmake_minimum_required(VERSION 3.25)
project(MetroidKWin LANGUAGES CXX)
set(CMAKE_AUTOMOC ON)
set(CMAKE_CXX_STANDARD 23)
find_package(Qt6 REQUIRED COMPONENTS Widgets DBus Quick)
find_package(KWin CONFIG REQUIRED)
# Some distributions omit this private header from the KWin development package.
# Supply the matching KWin source directory; never mix compositor versions.
set(KWIN_SOURCE_INCLUDE "" CACHE PATH "Directory containing the matching abstract_data_source.h")
add_library(metroid MODULE plugin.cpp)
target_include_directories(metroid PRIVATE ${KWIN_SOURCE_INCLUDE} /usr/include/kwin/wayland)
target_link_libraries(metroid PRIVATE KWin::kwin Qt6::Widgets Qt6::DBus)
set_target_properties(metroid PROPERTIES PREFIX "")
install(TARGETS metroid DESTINATION lib/qt6/plugins/kwin/plugins)
option(METROID_BUILD_ISOLATED_TEST_DRIVER "Build the test-only input module (never install)" OFF)
if(METROID_BUILD_ISOLATED_TEST_DRIVER)
    add_library(metroid-testinput MODULE ../../tests/kwin_input.cpp)
    target_link_libraries(metroid-testinput PRIVATE KWin::kwin Qt6::Widgets Qt6::DBus)
    set_target_properties(metroid-testinput PROPERTIES PREFIX "")
endif()
